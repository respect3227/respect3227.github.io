Shell 自动化脚本最佳实践

2026-08-02 · Shell · 石定睿

服务器巡检脚本

#!/bin/bash
echo "=== 磁盘 ===" && df -h
echo "=== 内存 ===" && free -h
echo "=== CPU ===" && uptime
echo "=== 端口 ===" && netstat -tlnp

日志清理脚本

#!/bin/bash
# 删除 7 天前的日志
find /var/log -name "*.log" -mtime +7 -delete
echo "清理完成: $(date)"
